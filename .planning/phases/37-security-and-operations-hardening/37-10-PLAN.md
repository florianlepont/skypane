    On the VPS (paste the single .pub line between the quotes):
    ```bash
    ssh ubuntu@<vps>
    sudo /opt/skypane/current/deploy/backup/install-backup-key.sh 'ssh-ed25519 AAAA... skypane-backup-pull@mac'

    sudo journalctl -u skypane-backup -n 20 --no-pager
    ls -l /var/lib/skypane-backup/archives/
    systemctl list-timers skypane-backup.timer --no-pager
    exit
    ```
    Back on the Mac:
    ```bash
    K="-i $HOME/.ssh/skypane_backup_ed25519 -o IdentitiesOnly=yes"
    ssh $K skypane-backup@<vps> list
    ssh $K skypane-backup@<vps> 'cat /opt/skypane/skypane.env'; echo "exit=$?"     # expect exit=2, no content
    ssh $K skypane-backup@<vps> 'get ../../../opt/skypane/skypane.env'; echo "exit=$?"   # expect exit=2
    ssh $K -t skypane-backup@<vps>; echo "exit=$?"                                     # expect no shell, exit=2
    ssh $K -N -L 9999:127.0.0.1:8643 skypane-backup@<vps>                               # expect port forwarding refused; Ctrl-C
    ```
